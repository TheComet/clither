\section{Snake Extrapolation}

\begin{equation}
    x(t) = a_0 + a_1t + \ldots + a_nt^n
\end{equation}

\begin{align*}
    x(t_0) &= x_0 = a_0 + a_1t_0 + \ldots + a_nt_0^n \\
    x(t_1) &= x_1 = a_0 + a_1t_1 + \ldots + a_nt_1^n \\
    \vdots & \\
    x(t_n) &= x_n = a_0 + a_1t_n + \ldots + a_nt_n^n
\end{align*}

\begin{align}
    \begin{bmatrix}
        x_0 \\ x_1 \\ \vdots \\ x_n
    \end{bmatrix}
    &=
    \begin{bmatrix}
        1      & t_0     & t_0^2     & \ldots & t_1^n \\
        1      & t_1     & t_1^2     & \ldots & t_2^n \\
        \vdots & \vdots  & \vdots    & \ddots & \vdots \\
        1      & t_n     & t_n^2     & \ldots & t_n^n
    \end{bmatrix}
    \begin{bmatrix}
        a_0 \\ a_1 \\ \vdots \\ a_n
    \end{bmatrix}
\end{align}

For a 3x3 matrix:

\begin{align}
    T^{-1} &= \begin{bmatrix}
        1 & 0 & 0 \\
        \frac{- t_{1} - t_{2}}{t_{1} t_{2}} & - \frac{t_{2}}{t_{1}^{2} - t_{1} t_{2}} & \frac{t_{1}}{t_{1} t_{2} - t_{2}^{2}}\\
        \frac{1}{t_{1} t_{2}} & \frac{1}{t_{1}^{2} - t_{1} t_{2}} & - \frac{1}{t_{1} t_{2} - t_{2}^{2}}
    \end{bmatrix} \\
    T^{-1} &= \begin{bmatrix}
        1 & 0 & 0 & 0 \\
        \frac{- t_{1} t_{2} - t_{1} t_{3} - t_{2} t_{3}}{t_{1} t_{2} t_{3}} &
        \frac{t_{2} t_{3}}{t_{1}^{3} - t_{1}^{2} t_{2} - t_{1}^{2} t_{3} + t_{1} t_{2} t_{3}} &
        -\frac{t_{1} t_{3}}{t_{1} t_{2}^{2} - t_{1} t_{2} t_{3} - t_{2}^{3} + t_{2}^{2} t_{3}} &
        \frac{t_{1} t_{2}}{t_{1} t_{2} t_{3} - t_{1} t_{3}^{2} - t_{2} t_{3}^{2} + t_{3}^{3}} \\
        \frac{t_{1} + t_{2} + t_{3}}{t_{1} t_{2} t_{3}} &
        \frac{- t_{2} - t_{3}}{t_{1}^{3} - t_{1}^{2} t_{2} - t_{1}^{2} t_{3} + t_{1} t_{2} t_{3}} &
        \frac{t_{1} + t_{3}}{t_{1} t_{2}^{2} - t_{1} t_{2} t_{3} - t_{2}^{3} + t_{2}^{2} t_{3}} &
        \frac{- t_{1} - t_{2}}{t_{1} t_{2} t_{3} - t_{1} t_{3}^{2} - t_{2} t_{3}^{2} + t_{3}^{3}} \\
        -\frac{1}{t_{1} t_{2} t_{3}} &
        \frac{1}{t_{1}^{3} - t_{1}^{2} t_{2} - t_{1}^{2} t_{3} + t_{1} t_{2} t_{3}} &
        -\frac{1}{t_{1} t_{2}^{2} - t_{1} t_{2} t_{3} - t_{2}^{3} + t_{2}^{2} t_{3}} &
        \frac{1}{t_{1} t_{2} t_{3} - t_{1} t_{3}^{2} - t_{2} t_{3}^{2} + t_{3}^{3}}
    \end{bmatrix}
\end{align}
