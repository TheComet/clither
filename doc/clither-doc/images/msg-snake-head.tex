\verb$MSG_SNAKE_HEAD$
\vspace{.25em}\\

\begin{bytefield}[bitwidth=3em,bitformatting=\small]{1}
    \bitheader{0-10} \\
    \bitbox{2}{frame}
    \bitbox{3}{pos x}
    \bitbox{3}{pos y}
    \bitbox{2}{angle}
    \bitbox{1}{speed}
\end{bytefield}

\begin{itemize}
    \item\textbf{frame:} Frame number of when the head was stepped.
    \item\textbf{pos x:} Q19.5 x coordinate of the snake's head position in world space.
    \item\textbf{pos y:} Q19.5 y coordinate of the snake's head position in world space.
    \item\textbf{angle:} Q4.12 angle of the snake's head.
    \item\textbf{speed:} Current speed of the snake's head.
\end{itemize}
