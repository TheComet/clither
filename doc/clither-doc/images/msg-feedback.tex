\verb$MSG_FEEDBACK$
\vspace{.25em}\\

\begin{bytefield}[bitwidth=3em,bitformatting=\small]{1}
    \bitheader{0-2} \\
    \bitbox{2}{frame}
    \bitbox{1}{diff}
\end{bytefield}

\begin{itemize}
    \item\textbf{frame:}   Server   frame-number  when  the   message   was   sent.
    \item\textbf{diff:}  Difference in simulation-ticks the server would  like  the
    client  to ``warp'' to. If negative, the client will speed  up  its  simulation
    speed, increasing the server-side command buffer. If  positive, the client will
    slow down its simulation speed.
\end{itemize}
