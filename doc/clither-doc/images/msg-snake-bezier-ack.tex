\verb$MSG_BEZIER$
\vspace{.25em}\\

\begin{bytefield}[bitwidth=3em,bitformatting=\small]{1}
    \bitheader{0-3} \\
    \bitbox{2}{snake ID}
    \bitbox{2}{handle ID}
\end{bytefield}

\begin{itemize}
    \item\textbf{snake ID:} Unique snake identifier.
    \item\textbf{handle ID:} Handle position in the ring buffer.
\end{itemize}
